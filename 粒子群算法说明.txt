  w := strtoint(trim(vlst[2].Replace('PSOw=', '', []))) / 100.0; // 惯性系数w：
  we := strtoint(trim(vlst[3].Replace('PSOwe=', '', []))) / 100.0; // 惯性因子最小值we:
  Vv := strtoint(trim(vlst[4].Replace('PSOVv=', '', []))) / 100.0; // 认知系数：
  c1 := strtoint(trim(vlst[5].Replace('PSOc1=', '', []))) / 100.0; // 个体学习因子c1：

  LOW := we;

  setlength(popu_copy, M, cityNum);

function TPSO.init: integer;
var
  i, j, k, tij: integer;
  rij: double;
  line: string;
  www, qq: Tstringlist;
  s, temps: string;
begin

  setlength(x, cityNum);
  setlength(y, cityNum);
  setlength(z, cityNum);

  setlength(distance, cityNum, cityNum);

  for i := 0 to cityNum - 1 do
  begin
    x[i] := trunc(MainForm.Display.Controller.Cities[i].x);
    y[i] := trunc(MainForm.Display.Controller.Cities[i].y);
    z[i] := (MainForm.Display.Controller.Cities[i].z);
  end;

  for i := 0 to cityNum - 1 do
  begin
    distance[i][i] := 0; // 对角线为0
    MainForm.Display.Controller.curdistance[i][i] := 0;
    for j := i + 1 to cityNum - 1 do
    begin
      rij := sqrt(((x[i] - x[j]) * (x[i] - x[j]) + (y[i] - y[j]) * (y[i] - y[j])
        ) / 1.0) * MainForm.Display.Controller.EdgeWeight[i][j];
      if MainForm.Display.Controller.distance[i][j] = -1.0 then
      begin
        distance[i][j] := rij + z[i];
        distance[j][i] := rij + z[j];
        MainForm.Display.Controller.curdistance[i][j] := distance[i][j];
        MainForm.Display.Controller.curdistance[j][i] := distance[i][j];

      end
      else
      begin
        distance[i][j] := MainForm.Display.Controller.distance[i][j] + z[i];
        // 不是 代替SQRT，否则仍然要乘以权数，自定义边长则不再乘权数乎
        distance[j][i] := MainForm.Display.Controller.distance[i][j] + z[j];
        MainForm.Display.Controller.curdistance[i][j] := distance[i][j];
        MainForm.Display.Controller.curdistance[j][i] := distance[i][j];
      end;
    end;
  end;
  distance[cityNum - 1][cityNum - 1] := 0;
  MainForm.Display.Controller.curdistance[cityNum - 1][cityNum - 1] := 0;
  setlength(bestTour, cityNum);
  

end;

function TPSO.PSO: integer;
var
  g, i, k: integer;
  rt: double;
  www: TP3;
begin
  for g := 1 to T do
  begin
    btupdate(); // 更新最优解

    for i := 0 to M - 1 do
      for k := 0 to cityNum - 1 do
        popu_copy[i][k] := popu[i][k];
    // 赋给中间变量取交叉变异
    ppupdate();
    // 交叉变异
    for i := 0 to M - 1 do
    begin
      fcopy[i] := fitness_alllength('popu_copy', i);
      // fcopy[i]=fitness(popu_copy[i]);
    end;
    // 计算交叉变异后的适应度
    // 比较并更新位置
    for i := 0 to M - 1 do
      if (fcopy[i] < fit[i]) then
      begin
        for k := 0 to cityNum - 1 do
          popu[i][k] := popu_copy[i][k];
        fit[i] := fcopy[i];
      end
      else
      begin
        rt := Random() / RM;
        if (rt <= AC) then
        begin
          for k := 0 to cityNum - 1 do
            popu[i][k] := popu_copy[i][k];
          fit[i] := fcopy[i];
        end;
      end;

  end;
end;

procedure TPSO.Rand_popu;
var
  i, j, k: integer;
begin
  for i := 0 to cityNum - 1 do
    allowedCities.Add(inttostr(i));
  for i := 0 to M - 1 do
  begin
    random_shuffle(cityNum - 1, allowedCities); // 随机生成全排列
    for k := 0 to cityNum - 1 do
      popu[i][k] := strtoint(allowedCities[k]);
    fit[i] := fitness_alllength('popu', i); // 计算生成解的适应值
  end;
end;

function TPSO.dis(ax, ay, bx, by: integer): double;
begin
  result := sqrt(power(ax - bx, 2) + power(ay - by, 2));
end;

function TPSO.remove_1(v: integer): integer;
begin
  if v = -1 then
    v := 0;
  result := v;

end;

function TPSO.fitness_alllength(cc: string; i: integer): double;
var
  sum: double;
  j: integer;
begin
  sum := 0;
  if (cc = 'popu') then
  begin
    for j := 0 to cityNum - 2 do
      sum := sum + distance[remove_1(popu[i][j])][remove_1(popu[i][j + 1])];
    sum := sum + distance[remove_1(popu[i][cityNum - 1])][remove_1(popu[i][0])];
  end;
  if (cc = 'popu_copy') then
  begin
    for j := 0 to cityNum - 2 do
      sum := sum + distance[remove_1(popu_copy[i][j])][remove_1(popu_copy[i][j + 1])];
    sum := sum + distance[remove_1(popu_copy[i][cityNum - 1])
      ][remove_1(popu_copy[i][0])];
  end;

    result := sum;

end;


procedure TPSO.random_shuffle(PSONum0: integer; allowedCities0: Tstringlist);
var
  i, j: integer;
  k: string;
begin
  i := Random(PSONum0);
  j := i;
  while i = j do
    j := Random(PSONum0);
  k := allowedCities0[i];
  allowedCities0[i] := allowedCities0[j];
  allowedCities0[j] := k;

end;

procedure TPSO.ppupdate;
begin
  if (c1 <> 0.0) then
    CroPb();
  if (c2 <> 0.0) then
    CroGb();
  // 这里是因为若有一个等于零则表示不学习
  mutation();
end;

procedure TPSO.btupdate;
var
  i, j, k: integer;
begin
  for i := 0 to M - 1 do
    if (fit[i] < pbest[i]) then
    begin
      pbest[i] := fit[i];
      for k := 0 to cityNum - 1 do
        Pbest_popu[i][k] := popu[i][k];
      if (pbest[i] < gbest) then
      begin
        gbest := pbest[i];
        for k := 0 to cityNum - 1 do
          Gbest_popu[k] := popu[i][k];
      end;
    end;
end;

procedure TPSO.CroPb;
var
  i, j, k: integer;
  r1, CroRate, rt: double;
begin
  for i := 0 to M - 1 do
  begin
    r1 := Random() / RM * (UP - LOW) + LOW;
    CroRate := c1 * r1;
    rt := Random() / RM;
    if (CroRate >= rt) then
      Cross(popu_copy[i], Pbest_popu[i], i, 'Pbest_popu');
  end;
end;

procedure TPSO.CroGb;
var
  i, j, k: integer;
  r2, CroRate, rt: double;
begin
  for i := 0 to M - 1 do
  begin
    r2 := Random() / RM * (UP - LOW) + LOW;
    CroRate := c2 * r2;
    rt := Random() / RM;
    if (CroRate >= rt) then
      Cross(popu_copy[i], Gbest_popu, i, 'Gbest_popu');
  end;
end;

procedure TPSO.mutation; // 随机变异
var
  MRate, r: double;
  i, p1, p2, www: integer;
begin

  MRate := w;
  for i := 0 to M - 1 do
  begin
    r := Random() / RM;
    if (r <= MRate) then
    begin
      p1 := Random(cityNum - 1) mod (cityNum - 1);
      p2 := Random(cityNum - 1) mod (cityNum - 1);
      while (p1 = p2) do
        p2 := Random(cityNum - 1) mod (cityNum - 1);
      // swap(popu_copy[i][p1],popu_copy[i][p2]);
      // swap操作实现交换两个容器内所有元素的功能
      www := popu_copy[i][p1];
      popu_copy[i][p1] := popu_copy[i][p2];
      popu_copy[i][p2] := www;
    end;
  end;

end;
